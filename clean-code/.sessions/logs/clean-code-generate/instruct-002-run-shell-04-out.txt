from __future__ import annotations

import inspect
from pathlib import Path
from typing import Any

from agents.action import action
from primitives.declared_operation import DeclaredOperation
from primitives.declared_property import DeclaredProperty
from primitives.instruction import Instruction
from scanners import ScannerCollection
from tools.tool import Toolset, resource, tool


class Generator(Toolset):
    """§ Purpose"""

    _declared_properties: dict[str, DeclaredProperty] = {
        "concepts": DeclaredProperty(
            "concepts",
            default_root="concepts/",
            key_discovery="fileStems",
        ),
        "examples": DeclaredProperty(
            "examples",
            default_root="examples",
            key_discovery="none",
        ),
        "formats": DeclaredProperty(
            "formats",
            default_root="formats/",
            key_discovery="subfolderNames",
        ),
        "template": DeclaredProperty(
            "template",
            default_root="formats/{activeKey}/{domain-slug}-template",
            key_discovery="none",
            active_key="format",
        ),
    }
    _declared_operations: dict[str, DeclaredOperation] = {
        "generate_output": DeclaredOperation("generate_output", target="generate_code"),
    }

    def __init__(self) -> None:
        super().__init__()

    @property
    def module_dir(self) -> Path:
        return Path(inspect.getfile(type(self))).resolve().parent

    @property
    def domain_slug(self) -> str:
        return self.module_dir.name

    def default_for(self, name: str) -> Instruction:
        declared = self._declared_properties.get(name)
        if declared is None:
            raise KeyError(name)
        return declared.route(self)

    def _scanner_root(self) -> Path:
        declared = self._declared_properties.get("formats")
        if declared and self._has_formats_folder():
            active = self._active_format()
            if active:
                return self.module_dir / "formats" / active / "scanners"
        return self.module_dir / "scanners"

    def _scanner_collection(self) -> ScannerCollection:
        return ScannerCollection(self.module_dir, self._scanner_root())

    def _has_formats_folder(self) -> bool:
        return (self.module_dir / "formats").is_dir()

    def _active_format(self) -> str | None:
        getter = getattr(self, "_declared_properties", {}).get("template")
        if getter is None or not getter.active_key:
            return None
        resource = getattr(self, getter.active_key, None)
        if callable(resource):
            resource = resource()
        if resource:
            return str(resource)
        return None

    def _instruction(self, text: str) -> Instruction:
        return Instruction(text, self.module_dir, domain_slug=self.domain_slug)

    def _expand_section_doc(self, section: str) -> str:
        return self._instruction(f"§ {section}").expand()

    def concepts(self) -> str:
        return self.default_for("concepts").expand()

    def examples(self) -> str:
        return self.default_for("examples").expand()

    def formats(self) -> str:
        declared = self._declared_properties["formats"]
        if not self._has_formats_folder():
            return ""
        keys = declared.discover_keys(self) or []
        return "\n".join(keys)

    def template(self) -> str:
        declared = self._declared_properties["template"]
        if self._has_formats_folder():
            active = self._active_format()
            if active:
                pattern = f"formats/{active}/{self.domain_slug}-template"
                for path in sorted((self.module_dir / "formats" / active).glob(f"{self.domain_slug}-template.*")):
                    return self._instruction(str(path.relative_to(self.module_dir)).replace("\\", "/")).expand()
                return self._instruction(pattern).expand()
        for path in sorted(self.module_dir.glob(f"{self.domain_slug}-template.*")):
            return self._instruction(path.name).expand()
        return self._instruction(f"{self.domain_slug}-template").expand()

    def generate_output(self) -> str:
        operation = self._declared_operations["generate_output"].route(self)
        if operation is None:
            return ""
        doc = inspect.getdoc(operation) or ""
        return doc.strip()

    @action
    def generate(self) -> str:
        """§ Generate"""
        self.concepts()
        self.examples()
        self.formats()
        self.template()
        self.generate_output()
        return "When done, run validate."

    @action
    def validate(self) -> str:
        """§ Validate"""
        self.concepts()
        self.scan()
        return "Validation report."

    @action
    def satisfy(self) -> str:
        """§ Satisfy"""
        self.concepts()
        self.template()
        return "When done, run validate."

    @tool
    def scanners(self) -> str:
        """Return the registered scanner rule catalog."""
        return self._scanner_collection().catalog()

    @tool
    def scan(self, paths: list[str]) -> str:
        """§ Scan"""
        files = [Path(path) for path in paths]
        report = self._scanner_collection().run(Path.cwd(), files)
        return str(report.to_dict())
